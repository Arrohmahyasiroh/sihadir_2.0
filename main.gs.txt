/**
 * Fungsi darurat: reset Settings koordinat jika koordinat sudah rusak/salah.
 * Jalankan dari GAS Editor jika absensi tidak bisa dilakukan karena koordinat error.
 */
function resetCoordinates() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sh = ss.getSheetByName(SHEET_NAMES.SETTINGS);
  if (!sh) { Logger.log('Sheet Settings tidak ditemukan!'); return; }

  const data = sh.getDataRange().getValues();
  const coordKeys = {
    'geo1_lat': '-7.250445',
    'geo1_lng': '112.768845',
    'geo1_radius': '100',
    'geo2_lat': '-7.251200',
    'geo2_lng': '112.769500',
    'geo2_radius': '80'
  };

  for (let i = 1; i < data.length; i++) {
    const key = data[i][0];
    if (coordKeys[key] !== undefined) {
      // Simpan sebagai STRING plain, bukan angka, agar Sheets tidak auto-convert
      sh.getRange(i + 1, 2).setNumberFormat('@').setValue(coordKeys[key]);
      Logger.log('Reset: ' + key + ' = ' + coordKeys[key]);
    }
  }

  try { SpreadsheetApp.getUi().alert('Koordinat berhasil direset.'); } catch (e) { Logger.log('Koordinat berhasil direset. (UI alert dilewati: ' + e.message + ')'); }
}

function generateID(prefix) {
  return prefix + '_' + new Date().getTime() + '_' + Math.floor(Math.random() * 1000);
}

// ─── ROLE PEGAWAI (GURU/TENDIK) ───────────────────────────
// Role yang valid untuk pegawai (bukan admin): 'user_guru' dan 'user_tendik'.
// Dipakai saat admin menyimpan Role lewat form, supaya tidak ikut tergeneralisasi
// jadi 'user' (bug lama: pilihan Guru/Tendik di form tidak pernah benar-benar tersimpan).
const ROLE_PEGAWAI_VALID = ['user_guru', 'user_tendik'];
function normalRolePegawai_(r) {
  return ROLE_PEGAWAI_VALID.includes(r) ? r : 'user_guru';
}
/**
 * Semua akun selain admin dihitung sebagai pegawai (guru/tendik) di dashboard & rekap —
 * termasuk akun lama yang masih berrole 'user' generik dari sebelum ada pembedaan Guru/Tendik.
 * Sengaja TIDAK membandingkan persis ke 'user_guru'/'user_tendik' supaya tidak ada akun yang
 * "hilang" dari dashboard/rekap hanya karena belum pernah diedit ulang rolenya.
 */
function isPegawaiAktif_(u) {
  return u.Role !== 'admin' && (u.Aktif === true || u.Aktif === 'TRUE');
}

function getSheetData(sheetName) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(sheetName);
  if (!sheet) return [];
  const data = sheet.getDataRange().getValues();
  if (data.length < 2) return [];
  const headers = data[0];
  return data.slice(1).map(row => {
    const obj = {};
    headers.forEach((h, i) => {
      let val = row[i];
      if (val instanceof Date && !isNaN(val)) val = val.toISOString();
      obj[h] = val;
    });
    return obj;
  });
}

function addLog(userID, nama, aksi, detail) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sh = ss.getSheetByName(SHEET_NAMES.LOG);
    if (!sh) return;
    sh.appendRow([generateID('LOG'), new Date(), userID, nama, aksi, detail, '']);
  } catch(e) {}
}

function getSettings() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sh = ss.getSheetByName(SHEET_NAMES.SETTINGS);
  if (!sh) return {};
  // Baca sebagai teks mentah (getDisplayValues) agar tidak auto-convert ke Date
  const raw   = sh.getDataRange().getDisplayValues(); // string asli seperti di cell
  const typed = sh.getDataRange().getValues();        // untuk deteksi Date jam_masuk/jam_pulang
  const cfg = {};
  for (let i = 1; i < raw.length; i++) {
    const key = raw[i][0];
    if (!key) continue;
    let val = typed[i][1];

    if (val instanceof Date && !isNaN(val)) {
      // Ini memang kolom waktu (jam_masuk / jam_pulang) → format HH:mm
      val = String(val.getHours()).padStart(2,'0') + ':' + String(val.getMinutes()).padStart(2,'0');
    } else {
      // Untuk koordinat & radius: ambil display value (teks) lalu parse
      // Ini menghindari Sheets salah mengubah "-7.250445" jadi serial Date
      val = raw[i][1];
    }
    cfg[key] = val;
  }
  return cfg;
}

/**
 * Validasi nilai settings koordinat. Kembalikan error string jika ada masalah.
 * Dipanggil di absenMasuk/absenPulang untuk debug lebih jelas.
 */

function validateGeoSettings(cfg) {
  const errors = [];
  ['geo1_lat','geo1_lng','geo1_radius','geo2_lat','geo2_lng','geo2_radius'].forEach(k => {
    const v = parseFloat(cfg[k]);
    if (isNaN(v)) errors.push(k + '="' + cfg[k] + '" tidak valid');
  });
  return errors;
}

/**
 * Test geofencing dengan koordinat tertentu.
 * Ganti lat/lng di bawah lalu jalankan dari GAS Editor.
 */

function testGeofencing() {
  const testLat = -7.250445; // ← ganti dengan koordinat HP kamu
  const testLng = 112.768845;

  const cfg = getSettings();
  const geo1Lat = parseFloat(cfg.geo1_lat);
  const geo1Lng = parseFloat(cfg.geo1_lng);

  const d2 = hitungJarak(testLat, testLng, geo2Lat, geo2Lng);

  Logger.log('Titik test: ' + testLat + ', ' + testLng);
  Logger.log(cfg.geo1_nama + ': center=' + geo1Lat + ',' + geo1Lng + ' radius=' + geo1R + 'm → jarak=' + Math.round(d1) + 'm → ' + (d1 <= geo1R ? 'DALAM ✅' : 'LUAR ❌'));
  Logger.log(cfg.geo2_nama + ': center=' + geo2Lat + ',' + geo2Lng + ' radius=' + geo2R + 'm → jarak=' + Math.round(d2) + 'm → ' + (d2 <= geo2R ? 'DALAM ✅' : 'LUAR ❌'));
}

// ─── KALKULASI JARAK HAVERSINE ────────────────────────────

// ─── KALKULASI JARAK HAVERSINE ────────────────────────────
function hitungJarak(lat1, lng1, lat2, lng2) {
  const R = 6371000;
  const dLat = (lat2 - lat1) * Math.PI / 180;
  const dLng = (lng2 - lng1) * Math.PI / 180;
  const a = Math.sin(dLat/2)**2 + Math.cos(lat1*Math.PI/180) * Math.cos(lat2*Math.PI/180) * Math.sin(dLng/2)**2;
  return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
}

// ─── [BARU] HITUNG DURASI JAM MENGAJAR ───────────────────
/**
 * Ubah jam ("HH:MM", string ISO, atau Date) menjadi menit sejak 00:00.
 * Return 0 jika kosong / tidak valid.
 */
function jamKeMenit(s) {
  if (s === null || s === undefined || s === '') return 0;
  if (s instanceof Date) return isNaN(s) ? 0 : s.getHours() * 60 + s.getMinutes();
  s = String(s);
  if (s.includes('T')) {
    const d = new Date(s);
    return isNaN(d) ? 0 : d.getHours() * 60 + d.getMinutes();
  }
  const parts = s.split(':');
  if (parts.length < 2) return 0;
  return (parseInt(parts[0]) || 0) * 60 + (parseInt(parts[1]) || 0);
}

/** Menit sejak 00:00 → "HH:MM". */
function menitKeStr(menit) {
  const p = n => String(n).padStart(2, '0');
  return p(Math.floor(menit / 60)) + ':' + p(menit % 60);
}

// ─── BATAS WAKTU ISI LAPORAN MENGAJAR & KEGIATAN ──────────
// Setelah (jam_pulang + batas_laporan_menit), menu Laporan Mengajar & Kegiatan
// otomatis terkunci untuk hari itu. Admin atur "Batas Waktu Isi Laporan" di
// Setting Geofencing; default 20 menit bila belum pernah diisi.
const BATAS_LAPORAN_MENIT_DEFAULT = 20;

function getBatasLaporanMenit_() {
  const v = parseFloat(getSettings().batas_laporan_menit);
  return v >= 0 ? v : BATAS_LAPORAN_MENIT_DEFAULT;
}

/**
 * { lewat, jamKunci } — lewat = waktu sekarang (server) sudah melewati jam_pulang
 * + batas; jamKunci = jam kuncinya dalam format "HH:MM" untuk ditampilkan ke user.
 */
function statusKunciWaktu_() {
  const cfg = getSettings();
  const [jh, jm] = String(cfg.jam_pulang || '16:00').split(':').map(n => parseInt(n) || 0);
  const batas = getBatasLaporanMenit_();
  const totalMenit = ((jh * 60 + jm + batas) % 1440 + 1440) % 1440; // dibungkus 24 jam, jaga-jaga jam_pulang larut malam
  const now = new Date();
  const batasWaktu = new Date(now);
  batasWaktu.setHours(jh, jm + batas, 0, 0); // Date otomatis menormalkan menit berlebih (termasuk lewat tengah malam)
  return { lewat: now > batasWaktu, jamKunci: menitKeStr(totalMenit) };
}

/**
 * Hitung durasi dalam menit dari string "HH:MM" ke "HH:MM".
 * Return 0 jika format tidak valid atau selisih negatif.
 */
function hitungDurasiMenit(jamMulai, jamSelesai) {
  if (!jamMulai || !jamSelesai) return 0;
  const durasi = jamKeMenit(jamSelesai) - jamKeMenit(jamMulai);
  return durasi > 0 ? durasi : 0;
}

// ─── SATUAN JAM PELAJARAN (JP) ────────────────────────────
// Kebijakan tetap sekolah ini: 1 JP = 30 menit. Jadi 07:30-08:30 (60 menit
// mengajar) dihitung 2 JP, bukan 1 jam. Dipakai di Rekap Jam Mengajar.
const MENIT_PER_JP = 30;

/** Menit → jumlah JP (pecahan ditampilkan apa adanya, dibulatkan maksimal 2 desimal). */
function menitKeJP(menit, perJP) {
  perJP = perJP || MENIT_PER_JP;
  return Math.round(menit / perJP * 100) / 100;
}

/** Menit → teks "2 JP" / "1,5 JP". */
function formatJP(menit, perJP) {
  return String(menitKeJP(menit, perJP)).replace('.', ',') + ' JP';
}
