// ─── ABSENSI MASUK ────────────────────────────────────────
function absenMasuk(token, lat, lng) {
  try {
    // [KEAMANAN] Validasi session — userID ambil dari server, bukan dari client
    const session = requireUser(token);
    const userID = session.userID;

    const users = getSheetData(SHEET_NAMES.USER);
    const user = users.find(u => u.ID === userID);
    if (!user) return { success: false, message: 'User tidak ditemukan' };
    const nama = user.Nama;

    const cfg = getSettings();
    const now = new Date();
    const tanggal = Utilities.formatDate(now, 'Asia/Jakarta', 'yyyy-MM-dd');

    const absensi = getSheetData(SHEET_NAMES.ABSENSI);
    const sudahAbsen = absensi.find(a =>
      a.UserID === userID &&
      Utilities.formatDate(new Date(a.Tanggal), 'Asia/Jakarta', 'yyyy-MM-dd') === tanggal &&
      a.WaktuMasuk
    );
    if (sudahAbsen) return { success: false, message: 'Anda sudah absen masuk hari ini pukul ' + Utilities.formatDate(new Date(sudahAbsen.WaktuMasuk), 'Asia/Jakarta', 'HH:mm') };

    // Validasi koordinat
    lat = parseFloat(lat); lng = parseFloat(lng);
    if (isNaN(lat) || isNaN(lng)) return { success: false, message: 'Koordinat lokasi tidak valid' };

    // [FIX] Validasi settings koordinat sebelum dipakai — tanda minus harus terjaga
    const geoErrors = validateGeoSettings(cfg);
    if (geoErrors.length > 0) {
      return { success: false, message: 'Konfigurasi geofencing tidak valid. Hubungi admin.\nDetail: ' + geoErrors.join(', ') };
    }

    const geo1Lat = parseFloat(cfg.geo1_lat), geo1Lng = parseFloat(cfg.geo1_lng), geo1R = parseFloat(cfg.geo1_radius);
    const geo2Lat = parseFloat(cfg.geo2_lat), geo2Lng = parseFloat(cfg.geo2_lng), geo2R = parseFloat(cfg.geo2_radius);
    const d1 = hitungJarak(lat, lng, geo1Lat, geo1Lng);
    const d2 = hitungJarak(lat, lng, geo2Lat, geo2Lng);

    let lokasiNama = '', didalam = false;
    if (d1 <= geo1R) { lokasiNama = cfg.geo1_nama || 'Gedung Utama'; didalam = true; }
    else if (d2 <= geo2R) { lokasiNama = cfg.geo2_nama || 'Gedung Lab'; didalam = true; }

    if (!didalam) {
      return { success: false, message: `Anda berada di luar area absensi.\nJarak ke ${cfg.geo1_nama||'Gedung Utama'}: ${Math.round(d1)}m (radius ${geo1R}m)\nJarak ke ${cfg.geo2_nama||'Gedung Lab'}: ${Math.round(d2)}m (radius ${geo2R}m)\n\nPastikan Anda berada di dalam area sekolah atau hubungi admin untuk cek pengaturan koordinat.` };
    }

    const [jH, jM] = (cfg.jam_masuk || '07:00').split(':').map(Number);
    const toleransi = parseInt(cfg.toleransi_menit) || 15;
    const batasLambat = new Date(now); batasLambat.setHours(jH, jM + toleransi, 0, 0);
    const tepat = new Date(now); tepat.setHours(jH, jM, 0, 0);
    const statusMasuk = now <= tepat ? 'TEPAT WAKTU' : now <= batasLambat ? 'TOLERANSI' : 'TERLAMBAT';

    const id = generateID('ABS');
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sh = ss.getSheetByName(SHEET_NAMES.ABSENSI);
    sh.appendRow([id, userID, nama, now, now, '', lokasiNama, '', lat, lng, '', '', statusMasuk, '', '']);
    addLog(userID, nama, 'ABSEN_MASUK', `Lokasi: ${lokasiNama} | Status: ${statusMasuk}`);
    return { success: true, message: `Absen masuk berhasil!\nLokasi: ${lokasiNama}\nStatus: ${statusMasuk}\nWaktu: ${Utilities.formatDate(now,'Asia/Jakarta','HH:mm')}`, absID: id, status: statusMasuk };
  } catch(e) {
    return { success: false, message: e.message };
  }
}

// ─── ABSENSI PULANG ───────────────────────────────────────
function absenPulang(token, lat, lng) {
  try {
    const session = requireUser(token);
    const userID = session.userID;

    const users = getSheetData(SHEET_NAMES.USER);
    const user = users.find(u => u.ID === userID);
    if (!user) return { success: false, message: 'User tidak ditemukan' };
    const nama = user.Nama;

    lat = parseFloat(lat); lng = parseFloat(lng);
    if (isNaN(lat) || isNaN(lng)) return { success: false, message: 'Koordinat lokasi tidak valid' };

    const cfg = getSettings();
    const now = new Date();
    const tanggal = Utilities.formatDate(now, 'Asia/Jakarta', 'yyyy-MM-dd');

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sh = ss.getSheetByName(SHEET_NAMES.ABSENSI);
    const data = sh.getDataRange().getValues();

    let rowIdx = -1;
    for (let i = 1; i < data.length; i++) {
      if (data[i][1] === userID && Utilities.formatDate(new Date(data[i][3]),'Asia/Jakarta','yyyy-MM-dd') === tanggal && data[i][4] && !data[i][5]) {
        rowIdx = i + 1; break;
      }
    }
    if (rowIdx < 0) return { success: false, message: 'Anda belum absen masuk hari ini atau sudah absen pulang' };

    // Khusus role guru: wajib isi minimal 1 laporan mengajar hari ini sebelum bisa absen pulang.
    if (session.role === 'user_guru') {
      const sudahLaporan = getSheetData(SHEET_NAMES.LAPORAN).some(l =>
        l.UserID === userID && l.Tanggal &&
        Utilities.formatDate(new Date(l.Tanggal), 'Asia/Jakarta', 'yyyy-MM-dd') === tanggal
      );
      if (!sudahLaporan) return { success: false, message: 'Anda belum mengisi laporan mengajar hari ini. Isi Laporan Mengajar terlebih dahulu sebelum absen pulang.' };
    }

    // [FIX] Validasi settings koordinat
    const geoErrsPulang = validateGeoSettings(cfg);
    if (geoErrsPulang.length > 0) {
      return { success: false, message: 'Konfigurasi geofencing tidak valid. Hubungi admin.' };
    }
    const d1 = hitungJarak(lat, lng, parseFloat(cfg.geo1_lat), parseFloat(cfg.geo1_lng));
    const d2 = hitungJarak(lat, lng, parseFloat(cfg.geo2_lat), parseFloat(cfg.geo2_lng));
    let lokasiNama = '', didalam = false;
    if (d1 <= parseFloat(cfg.geo1_radius)) { lokasiNama = cfg.geo1_nama || 'Gedung Utama'; didalam = true; }
    else if (d2 <= parseFloat(cfg.geo2_radius)) { lokasiNama = cfg.geo2_nama || 'Gedung Lab'; didalam = true; }
    if (!didalam) return { success: false, message: `Anda di luar area absensi.\nJarak ke ${cfg.geo1_nama}: ${Math.round(d1)}m (radius ${parseFloat(cfg.geo1_radius)}m)\nJarak ke ${cfg.geo2_nama}: ${Math.round(d2)}m (radius ${parseFloat(cfg.geo2_radius)}m)` };

    const [jH, jM] = (cfg.jam_pulang || '16:00').split(':').map(Number);
    const batasPulang = new Date(now); batasPulang.setHours(jH, jM, 0, 0);
    const statusPulang = now >= batasPulang ? 'NORMAL' : 'PULANG AWAL';

    sh.getRange(rowIdx, 6).setValue(now);
    sh.getRange(rowIdx, 8).setValue(lokasiNama);
    sh.getRange(rowIdx, 11).setValue(lat);
    sh.getRange(rowIdx, 12).setValue(lng);
    sh.getRange(rowIdx, 14).setValue(statusPulang);
    addLog(userID, nama, 'ABSEN_PULANG', `Lokasi: ${lokasiNama} | Status: ${statusPulang}`);
    return { success: true, message: `Absen pulang berhasil!\nLokasi: ${lokasiNama}\nStatus: ${statusPulang}`, status: statusPulang };
  } catch(e) {
    return { success: false, message: e.message };
  }
}

// ─── CEK STATUS ABSEN HARI INI ────────────────────────────
function cekStatusAbsenHariIni(token) {
  try {
    const session = requireUser(token);
    const userID = session.userID;
    const now = new Date();
    const tanggal = Utilities.formatDate(now, 'Asia/Jakarta', 'yyyy-MM-dd');
    const absensi = getSheetData(SHEET_NAMES.ABSENSI);
    // [FIX] Jangan panggil formatDate untuk Tanggal kosong/rusak — satu baris kotor
    // bikin seluruh fungsi throw dan dashboard user balik error (halaman kosong).
    const abs = absensi.find(a => {
      if (a.UserID !== userID || !a.Tanggal) return false;
      const d = new Date(a.Tanggal);
      return !isNaN(d) && Utilities.formatDate(d, 'Asia/Jakarta', 'yyyy-MM-dd') === tanggal;
    });
    if (!abs) return { masuk: false, pulang: false };
    return {
      masuk: !!abs.WaktuMasuk,
      pulang: !!abs.WaktuPulang,
      waktuMasuk: abs.WaktuMasuk ? Utilities.formatDate(new Date(abs.WaktuMasuk),'Asia/Jakarta','HH:mm') : null,
      waktuPulang: abs.WaktuPulang ? Utilities.formatDate(new Date(abs.WaktuPulang),'Asia/Jakarta','HH:mm') : null,
      statusMasuk: abs.StatusMasuk,
      lokasi: abs.LokasiMasuk
    };
  } catch(e) { return { masuk: false, pulang: false, error: e.message }; }
}
